%%% PREAMBLE
% Open Sans (loaded by elife.cls) has no scalable Type 1 metrics, so
% microtype's font expansion can't be applied under pdflatex. Disable it
% before elife.cls loads microtype.
\PassOptionsToPackage{expansion=false}{microtype}
\documentclass[$if(fontsize)$$fontsize$$else$9pt$endif$$if(linenumbers)$,lineno$endif$$if(linespacing)$,$linespacing$$endif$]{elife}

%%% Packages preserved from the original eLife template
\usepackage[version=4]{mhchem}
\usepackage{siunitx}
\DeclareSIUnit\Molar{M}

%%% Pandoc/Quarto scaffolding
\providecommand{\tightlist}{%
  \setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}
\usepackage{graphicx}
% Pandoc's own \pandocbounded: an image wider than the line or taller than the page is scaled down to fit.
% The identity the scaffolding used to provide let a figure laid out for 180 mm run off the page edge.
\makeatletter
\newsavebox\pandoc@box
\newcommand*\pandocbounded[1]{%
  \sbox\pandoc@box{#1}%
  \Gscale@div\@tempa{0.92\textheight}{\dimexpr\ht\pandoc@box+\dp\pandoc@box\relax}%
  \Gscale@div\@tempb{\linewidth}{\wd\pandoc@box}%
  \ifdim\@tempb\p@<\@tempa\p@\let\@tempa\@tempb\fi
  \ifdim\@tempa\p@<\p@\scalebox{\@tempa}{\usebox\pandoc@box}\else\usebox{\pandoc@box}\fi}
\makeatother
% Figures span the page as in eLife's own template, 4.5 cm into the left margin (188 mm in all), so a figure
% laid out for 180 mm prints at that size: a figure float's box is set that wide and then shifted left.
\makeatletter
\def\elife@figname{figure}
\newcommand\elife@figshift{4.5cm}
\apptocmd{\@floatboxreset}{\ifx\@captype\elife@figname\hsize=\dimexpr\textwidth+\elife@figshift\relax\linewidth=\hsize\columnwidth=\hsize\fi}{}{\GenericError{}{elife: cannot patch floatboxreset}{}{}}
\apptocmd{\@endfloatbox}{\ifx\@captype\elife@figname\global\setbox\@currbox\vbox{\moveleft\elife@figshift\box\@currbox}\fi}{}{\GenericError{}{elife: cannot patch endfloatbox}{}{}}
\makeatother
\usepackage{longtable,booktabs,array}
\usepackage{calc}
\usepackage{etoolbox}
% pandoc emits \def\LTcaptype{none} for un-captioned longtables; the caption
% package (loaded by elife.cls) then asks for a counter named "none". Define it.
\makeatletter
\@ifundefined{c@none}{\newcounter{none}}{}
\makeatother
% pandoc passes Shaded/Highlighting macros when code blocks are present
$if(highlighting-macros)$
$highlighting-macros$
$endif$

% Quarto cross-reference scaffolding (no-ops if quarto-scos isn't used)
\providecommand{\citeproctext}{}
\providecommand{\citeproc}[2]{\citep{#1}}

$for(header-includes)$
$header-includes$
$endfor$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%% ARTICLE SETUP
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\title{$title$}

$for(by-author)$
\author[$for(it.affiliations)$$it.number$$sep$,$endfor$$if(it.attributes.corresponding)$*$endif$]{$it.name.literal$}
$endfor$

$for(by-affiliation)$
\affil[$it.number$]{$it.name$$if(it.department)$, $it.department$$endif$$if(it.city)$, $it.city$$endif$$if(it.country)$, $it.country$$endif$}
$endfor$

$for(by-author)$$if(it.attributes.corresponding)$\corr{$it.email$}{$it.name.literal$}
$endif$$endfor$

$for(by-author)$$if(it.attributes.equal-contributor)$\contrib[\authfn{1}]{These authors contributed equally to this work.}
$endif$$endfor$

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%%% ARTICLE START
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{document}

\maketitle

$if(abstract)$
\begin{abstract}
$abstract$
\end{abstract}
$endif$

$if(impact-statement)$
\noindent\textbf{Impact statement.}\space $impact-statement$\par\medskip
$endif$

$if(keywords)$
\noindent\textbf{Keywords:}\space $for(keywords)$$keywords$$sep$; $endfor$\par\medskip
$endif$

$body$

$if(natbib)$
$if(bibliography)$
\bibliography{$for(bibliography)$$bibliography$$sep$,$endfor$}
$endif$
$endif$
$if(biblatex)$
\printbibliography$if(biblio-title)$[title=$biblio-title$]$endif$
$endif$

\end{document}
